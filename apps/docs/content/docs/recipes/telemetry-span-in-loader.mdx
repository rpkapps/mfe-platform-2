---
title: Telemetry span in a loader
description: Time a route loader with a span that is automatically enriched with the MFE, instance and route.
---

```tsx title="src/routes/wells/$wellId.tsx"
import { createFileRoute, notFound } from "@tanstack/react-router"

export const Route = createFileRoute("/wells/$wellId")({
  loader: async ({ context, params, abortController }) => {
    const { telemetry, runtime } = context.platform
    const span = telemetry.span("load-well", { wellId: params.wellId })
    try {
      const response = await fetch(`${runtime.env.API_BASE_URL}/wells/${params.wellId}`, {
        signal: abortController.signal,
      })
      if (response.status === 404) {
        span.end({ outcome: "not-found" })
        throw notFound()
      }
      if (!response.ok) throw new Error(`Well request failed (${response.status})`)
      const well = await response.json()
      const child = span.child("parse")
      const parsed = wellSchema.parse(well)
      child.end()
      span.end({ outcome: "ok" })
      return parsed
    } catch (error) {
      span.fail(error)
      throw error
    }
  },
  staticData: { breadcrumb: { fromLoader: "name" } },
})
```

What the adapter receives (with a span-capable adapter, or as `span:load-well` events otherwise):

```json
{
  "span.name": "load-well",
  "span.duration_ms": 118,
  "wellId": "w01",
  "outcome": "ok",
  "mfeId": "well-planner",
  "instanceId": "well-planner#1",
  "route": "/wells/$wellId",
  "environment": "staging",
  "release": "3.4.0",
  "remoteRelease": "2.1.0",
  "sessionId": "…"
}
```

`span.fail` also reports the error through `adapter.error` with the same attributes. If the shell has no telemetry adapter, all of this is a no-op; if the adapter throws, the loader is unaffected.
