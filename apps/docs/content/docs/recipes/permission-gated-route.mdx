---
title: Permission-gated route
description: Gate a route on a permission group with a native guard, declare the group for preflight, and show a friendly denied state.
---

```tsx title="src/routes/wells/$wellId/edit.tsx"
import { createFileRoute, redirect } from "@tanstack/react-router"

export const Route = createFileRoute("/wells/$wellId/edit")({
  beforeLoad: ({ context, params }) => {
    const { permissions, telemetry } = context.platform
    if (!permissions.hasGroup("wells:write")) {
      telemetry.track("guard.denied", {
        route: "/wells/$wellId/edit",
        wellId: params.wellId,
      })
      throw redirect({
        to: "/wells/$wellId",
        params: { wellId: params.wellId },
        search: { denied: "wells:write" },
      })
    }
  },
  staticData: {
    breadcrumb: "Edit",
    permissionGroups: ["wells:write"], // manifest metadata: preflight + palette filtering
  },
  component: EditAssetPage,
})
```

```tsx title="src/routes/wells/$wellId.tsx (excerpt)"
import { z } from "zod"

export const Route = createFileRoute("/wells/$wellId")({
  validateSearch: z.object({ denied: z.string().optional() }),
  component: () => {
    const { denied } = Route.useSearch()
    return (
      <>
        {denied && (
          <Alert variant="warning">
            <AlertTitle>Read only</AlertTitle>
            <AlertDescription>You need the {denied} group to edit this well.</AlertDescription>
          </Alert>
        )}
        <AssetDetails />
      </>
    )
  },
})
```

Why this shape:

- The guard is TanStack's `beforeLoad`; nothing platform-specific decides.
- `staticData.permissionGroups` lets the shell hide the route from the palette and preflight the whole MFE if every route needs the group — it never replaces the guard.
- The denied state stays inside the MFE and is explained to the user.
- Backend authorization still applies to the save request.

Test it with a bridge without the group:

```ts
const bridge = createTestBridge({ mfeId: "well-planner", permissionGroups: ["wells:read"] })
const mounted = renderMfe(definition, { bridge, path: "/well-planner/wells/w01/edit" })
await waitFor(() => expect(mounted.location().search).toContain("denied=assets%3Awrite"))
```
